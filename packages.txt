# Khadi — the Arch packages the desktop needs.
#
# One name per line, `#` to end of line is a comment. Install with:
#
#     sudo pacman -S --needed $(sed 's/#.*//' packages.txt | tr -s '[:space:]' ' ')
#
# Arch does not split headers into -dev packages, so most of these are needed
# both to build and to run. The upstream project documents Debian names; these
# are the Arch equivalents, checked against what smithay actually links.

# --- build ---------------------------------------------------------------
rust               # cargo. rustup works too; the workspace is edition 2024,
                   # so a 2024-capable toolchain is the real requirement.
pkgconf            # how install.sh finds the libraries below

# --- what the compositor links against -----------------------------------
# smithay is built with backend_drm, backend_gbm, backend_egl, backend_udev,
# backend_libinput, backend_session_libseat and renderer_gl, and each of those
# is one of these.
libinput           # input devices
seatd              # libseat: taking the seat without being root. The seatd
                   # service is only needed without systemd-logind; on a
                   # systemd install libseat uses logind and seatd stays off.
mesa               # libgbm and EGL
systemd-libs       # libudev. Already present on any systemd install; named so
                   # the dependency is written down rather than assumed.
wayland
libxkbcommon

# --- what the session needs at runtime -----------------------------------
xorg-xwayland      # X11 applications. khadi-comp starts and manages it itself.
cage               # the last fallback in khadi-session: if khadi-comp cannot
                   # drive the display twice over, the shell runs inside this
                   # rather than leaving you at a black screen.
networkmanager     # nmcli, which the settings screen drives for Wi-Fi
bluez              # the Bluetooth stack
bluez-utils        # bluetoothctl, which the settings screen drives
pipewire           # audio
wireplumber        # wpctl, which the volume control calls
curl               # geolocation lookups for the globe
fontconfig         # fc-list, how the shell finds an installed Nerd Font

# --- portals -------------------------------------------------------------
# session/khadi-portals.conf asks for the gtk backend by name, and for
# gnome-keyring to answer the Secret portal. Without these a file chooser in
# Firefox hangs waiting for a backend that is not there.
xdg-desktop-portal
xdg-desktop-portal-gtk
gnome-keyring

# --- fonts ---------------------------------------------------------------
# Chakra Petch and JetBrains Mono are bundled in shell/assets/fonts, so the
# interface has its typefaces with nothing installed. This is for the prompt:
# the shell looks for an installed Nerd Font for powerline and icon glyphs,
# and renders tofu without one.
ttf-jetbrains-mono-nerd

# --- the lock and login screens ------------------------------------------
pam                # the lock screen asks PAM whether the password is right. Part
                   # of any Arch install already; named so the dependency is
                   # written down rather than assumed. The rules live in
                   # /etc/pam.d/khadi-lock, which install.sh puts there.

# --- screenshots and recording -------------------------------------------
wl-clipboard       # wl-copy, for putting a screenshot on the clipboard. Without
                   # it the picture is still saved; only copying is lost.
ffmpeg             # screen recording. Without it the shell says so rather than
                   # appearing to record nothing.

# --- boot splash ----------------------------------------------------------
# install-boot-splash.sh adds the theme and rebuilds the initramfs. Optional:
# nothing else needs it, and uninstall-boot-splash.sh puts the old one back.
plymouth

# --- first-boot setup -----------------------------------------------------
# The questions a new machine asks about itself, on a VT before anything
# graphical exists. See provisioning/setup-form.sh.
gum                # the prompts themselves
#
# NOT listed: tzupdate, which guesses the timezone from the public address. It
# is in the AUR and not the repos, and `pacman -S` installs nothing at all when
# one name on its command line cannot be resolved — so a single AUR package here
# would break the install line this file documents. The form copes without it:
# the timezone prompt becomes a search box rather than a list with the right
# answer already picked. Install it yourself if you want the guess.

# --- login ---------------------------------------------------------------
# NOT YET SETTLED. install.sh registers a session in /usr/share/wayland-sessions,
# which needs a display manager to offer it, and Arch ships none by default.
# Khadi had its own greeter on greetd; it was dropped with the old codebase and
# is in the history at the tag pre-khadi. Upstream lists a login screen under
# "not done yet", so this is a real gap for both projects.
greetd             # the display manager the old greeter ran under
